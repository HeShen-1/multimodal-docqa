## 快速开始

### 环境要求

- Node.js 20.0+
- npm 10.0+

### 安装步骤

**1. 克隆项目**
```bash
git clone https://github.com/yourname/multimodal-docqa.git
cd multimodal-docqa/frontend
```

**2. 安装依赖**
```bash
npm install
```

**3. 配置环境变量**
```bash
# .env.development
REACT_APP_API_BASE_URL=http://localhost:8000
REACT_APP_ENV=development
```

**4. 启动开发服务器**
```bash
npm start
```

**5. 访问应用**
```
http://localhost:3000
```

## 核心功能开发

### 1. 文档上传功能

**功能描述**: 支持拖拽上传和点击上传PDF/DOCX文档。

**组件**: `DocumentUpload`

**核心代码**:

```typescript
// components/DocumentUpload/DocumentUpload.tsx
import React, { useState, useCallback } from 'react';
import { documentService } from '@/services/documentService';
import { useToast } from '@/hooks/useToast';
import styles from './DocumentUpload.module.css';

interface DocumentUploadProps {
  onUploadSuccess?: (documentId: string) => void;
  onUploadError?: (error: Error) => void;
}

export const DocumentUpload: React.FC<DocumentUploadProps> = ({
  onUploadSuccess,
  onUploadError,
}) => {
  const [uploading, setUploading] = useState(false);
  const [progress, setProgress] = useState(0);
  const [dragActive, setDragActive] = useState(false);
  const { showToast } = useToast();

  const handleDrag = useCallback((e: React.DragEvent) => {
    e.preventDefault();
    e.stopPropagation();
    if (e.type === 'dragenter' || e.type === 'dragover') {
      setDragActive(true);
    } else if (e.type === 'dragleave') {
      setDragActive(false);
    }
  }, []);

  const handleDrop = useCallback(async (e: React.DragEvent) => {
    e.preventDefault();
    e.stopPropagation();
    setDragActive(false);

    const files = e.dataTransfer.files;
    if (files && files.length > 0) {
      await handleUpload(files[0]);
    }
  }, []);

  const handleFileSelect = useCallback(async (e: React.ChangeEvent<HTMLInputElement>) => {
    const files = e.target.files;
    if (files && files.length > 0) {
      await handleUpload(files[0]);
    }
  }, []);

  const handleUpload = async (file: File) => {
    // 验证文件类型
    const allowedTypes = ['application/pdf', 'application/vnd.openxmlformats-officedocument.wordprocessingml.document'];
    if (!allowedTypes.includes(file.type)) {
      showToast('仅支持PDF和DOCX格式', 'error');
      return;
    }

    // 验证文件大小（10MB）
    if (file.size > 10 * 1024 * 1024) {
      showToast('文件大小不能超过10MB', 'error');
      return;
    }

    try {
      setUploading(true);
      setProgress(0);

      const result = await documentService.upload(file, (progress) => {
        setProgress(progress);
      });

      showToast('上传成功', 'success');
      onUploadSuccess?.(result.documentId);
    } catch (error) {
      showToast('上传失败', 'error');
      onUploadError?.(error as Error);
    } finally {
      setUploading(false);
      setProgress(0);
    }
  };

  return (
    <div
      className={`${styles.uploadArea} ${dragActive ? styles.dragActive : ''}`}
      onDragEnter={handleDrag}
      onDragLeave={handleDrag}
      onDragOver={handleDrag}
      onDrop={handleDrop}
    >
      {uploading ? (
        <div className={styles.uploadProgress}>
          <div className={styles.progressBar}>
            <div className={styles.progressFill} style={{ width: `${progress}%` }} />
          </div>
          <p>上传中... {progress}%</p>
        </div>
      ) : (
        <>
          <div className={styles.uploadIcon}>📄</div>
          <p className={styles.uploadText}>拖拽文件到此处或点击上传</p>
          <p className={styles.uploadHint}>支持PDF、DOCX格式，最大10MB</p>
          <input
            type="file"
            accept=".pdf,.docx"
            onChange={handleFileSelect}
            className={styles.fileInput}
          />
        </>
      )}
    </div>
  );
};
```

### 2. 智能问答功能

**功能描述**: 输入问题，调用后端API，展示答案和推理链。

**组件**: `QueryInput` + `AnswerDisplay` + `ThinkingChain`

**核心代码**:

```typescript
// pages/Query/Query.tsx
import React, { useState } from 'react';
import { QueryInput } from '@/components/QueryInput';
import { AnswerDisplay } from '@/components/AnswerDisplay';
import { ThinkingChain } from '@/components/ThinkingChain';
import { queryService } from '@/services/queryService';
import { QueryResponse } from '@/types/query';
import styles from './Query.module.css';

export const Query: React.FC = () => {
  const [loading, setLoading] = useState(false);
  const [result, setResult] = useState<QueryResponse | null>(null);

  const handleQuery = async (question: string) => {
    try {
      setLoading(true);
      setResult(null);

      const response = await queryService.query({
        question,
        topK: 5,
        enableThinking: true,
      });

      setResult(response);
    } catch (error) {
      console.error('查询失败:', error);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className={styles.queryPage}>
      <div className={styles.querySection}>
        <h1>智能问答</h1>
        <QueryInput onSubmit={handleQuery} loading={loading} />
      </div>

      {result && (
        <div className={styles.resultSection}>
          {result.thinking && result.thinking.length > 0 && (
            <ThinkingChain steps={result.thinking} />
          )}
          <AnswerDisplay
            answer={result.answer}
            sources={result.sources}
            processingTime={result.processingTime}
          />
        </div>
      )}
    </div>
  );
};
```

**QueryInput组件**:

```typescript
// components/QueryInput/QueryInput.tsx
import React, { useState } from 'react';
import styles from './QueryInput.module.css';

interface QueryInputProps {
  onSubmit: (question: string) => void;
  loading?: boolean;
}

export const QueryInput: React.FC<QueryInputProps> = ({ onSubmit, loading }) => {
  const [question, setQuestion] = useState('');

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (question.trim() && !loading) {
      onSubmit(question.trim());
    }

  return (
    <form className={styles.queryForm} onSubmit={handleSubmit}>

        className={styles.queryInput}
        placeholder="请输入您的问题..."
        value={question}
        onChange={(e) => setQuestion(e.target.value)}
        rows={4}
        disabled={loading}
      />
      <button
        type="submit"
        className={styles.submitButton}
        disabled={!question.trim() || loading}
      >
        {loading ? '查询中...' : '提交问题'}
      </button>
    </form>
  );
};
```

### 3. 流式输出功能

**功能描述**: 使用SSE实时接收并展示生成的答案。

**核心代码**:

```typescript
// hooks/useStreamQuery.ts
import { useState, useCallback } from 'react';
import { queryService } from '@/services/queryService';
import { QueryRequest, ThinkingStep } from '@/types/query';

export const useStreamQuery = () => {
  const [answer, setAnswer] = useState('');
  const [thinking, setThinking] = useState<ThinkingStep[]>([]);
  const [loading, setLoading] = useState(false);

  const query = useCallback(async (request: QueryRequest) => {
    setLoading(true);
    setAnswer('');
    setThinking([]);

    await queryService.queryStream(
      request,
      (chunk) => {
        const data = JSON.parse(chunk);
        
        if (data.type === 'thinking') {
          setThinking((prev) => [...prev, { step: data.step, content: data.content }]);
        } else if (data.type === 'answer') {
          setAnswer((prev) => prev + data.content);
        }
      },
      () => {
        setLoading(false);
      },
      (error) => {
        console.error('流式查询失败:', error);
        setLoading(false);
      }
    );
  }, []);

  return { answer, thinking, loading, query };
};
```

**使用示例**:

```typescript
// pages/Query/Query.tsx
import { useStreamQuery } from '@/hooks/useStreamQuery';

export const Query: React.FC = () => {
  const { answer, thinking, loading, query } = useStreamQuery();

  const handleQuery = (question: string) => {
    query({
      question,
      topK: 5,
      enableThinking: true,
    });
  };

  return (
    <div>
      <QueryInput onSubmit={handleQuery} loading={loading} />
      {thinking.length > 0 && <ThinkingChain steps={thinking} />}
      {answer && <AnswerDisplay answer={answer} />}
    </div>
  );
};
```

## 组件开发指南

### 组件设计原则

1. **单一职责**: 每个组件只负责一个功能
2. **可复用性**: 通用组件放在`components/common/`
3. **Props类型化**: 使用TypeScript定义Props接口
4. **样式隔离**: 使用CSS Modules避免样式冲突

### 组件模板

```typescript

  title,
```

## 状态管理

### Context API使用

```typescript
// contexts/AppContext.tsx
import React, { createContext, useContext, useState, ReactNode } from 'react';
import { Document } from '@/types/document';

interface AppState {
  documents: Document[];
  currentQuery: string;
}

interface AppContextType {
  state: AppState;
  setDocuments: (documents: Document[]) => void;
  setCurrentQuery: (query: string) => void;
}

const AppContext = createContext<AppContextType | undefined>(undefined);

export const AppProvider: React.FC<{ children: ReactNode }> = ({ children }) => {
  const [state, setState] = useState<AppState>({
    documents: [],
    currentQuery: '',
  });

  const setDocuments = (documents: Document[]) => {
    setState((prev) => ({ ...prev, documents }));
  };

  const setCurrentQuery = (query: string) => {
    setState((prev) => ({ ...prev, currentQuery: query }));
  };

  return (
    <AppContext.Provider value={{ state, setDocuments, setCurrentQuery }}>
      {children}
    </AppContext.Provider>
  );
};

export const useApp = () => {
  const context = useContext(AppContext);
  if (!context) {
    throw new Error('useApp must be used within AppProvider');
  }
  return context;
};
```

## API集成

### Axios配置

```typescript
// services/api.ts
import axios, { AxiosInstance } from 'axios';

const API_BASE_URL = process.env.REACT_APP_API_BASE_URL || 'http://localhost:8000';

export const apiClient: AxiosInstance = axios.create({
  baseURL: API_BASE_URL,
  timeout: 60000,
  headers: {
    'Content-Type': 'application/json',
  },
});

// 请求拦截器
apiClient.interceptors.request.use(
  (config) => {
    const token = localStorage.getItem('access_token');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => Promise.reject(error)
);

// 响应拦截器
apiClient.interceptors.response.use(
  (response) => response.data,
  (error) => {
    if (error.response?.status === 401) {
      localStorage.removeItem('access_token');
      window.location.href = '/login';
    }
    return Promise.reject(error);
  }
);
```

### 服务层封装

```typescript
// services/queryService.ts
import { apiClient } from './api';
import { QueryRequest, QueryResponse } from '@/types/query';

export const queryService = {
  async query(request: QueryRequest): Promise<QueryResponse> {
    return apiClient.post('/api/v1/query', request);
  },

  async queryStream(
    request: QueryRequest,
    onChunk: (chunk: string) => void,
    onComplete?: () => void,
    onError?: (error: Error) => void
  ): Promise<void> {
    const API_BASE_URL = process.env.REACT_APP_API_BASE_URL;
    
    try {
      const response = await fetch(`${API_BASE_URL}/api/v1/query/stream`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${localStorage.getItem('access_token')}`,
        },
        body: JSON.stringify(request),
      });

      const reader = response.body?.getReader();
      const decoder = new TextDecoder();

      while (true) {
        const { done, value } = await reader!.read();
        if (done) {
          onComplete?.();
          break;
        }

        const chunk = decoder.decode(value);
        const lines = chunk.split('\n');

        for (const line of lines) {
          if (line.startsWith('data: ')) {
            onChunk(line.slice(6));
          }
        }
      }
    } catch (error) {
      onError?.(error as Error);
    }
  },
};
```
